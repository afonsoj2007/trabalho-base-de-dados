Esquema Entidade-Relacionamento (Tabelas do Sistema)

users

PK id (INT)

name, email (UNIQUE)

password_hash (bcrypt)

role ('admin' | 'client')

categories

PK id (INT)

name (UNIQUE), slug

description, icon

products

PK id (INT)

FK category_id → categories(id)

name, price (CHECK ≥ 0)

stock (CHECK ≥ 0), image_url

orders

PK id (INT)

FK user_id → users(id)

total, status (CHECK)

shipping_address, payment_method

order_items

PK id (INT)

FK order_id → orders(id)

FK product_id → products(id)

quantity (> 0), unit_price (≥ 0)

reviews

PK id (INT)

FK product_id → products(id)

FK user_id → users(id)

rating (1..5), UNIQUE(prod, user)
